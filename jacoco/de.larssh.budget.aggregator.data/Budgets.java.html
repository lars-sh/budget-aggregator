		for (final Sheet sheet : sheetsFile.getSheets()) {

				final Set<Budget> budget = Budget.of(sheet);

				// Add References
				sheetsFile.getFileName().ifPresent(fileName -> setReferenceIfAbsent(budget, BudgetReference.FILE_NAME, fileName));

				throw new StringParseException(e, "Failed reading sheet%s.", sheet.getName().map(name -> " \"" + name + '\"').orElse(""));
			}
		}
		return budgets;
	}

	public static void removeDuplicateBudgets(final List<Budget> budgets) {
		// Prerequisite: budgets must be sorted!
		for (int index = budgets.size() - 1; index > 0; index -= 1) {
			if (budgets.get(index).equalsIncludingBalances(budgets.get(index - 1))) {
				budgets.remove(index);
				index -= 1;
			}
		}
	}

		// Get all accounts
		final Set<Account> accounts =  //

		for (final Account account : accounts) {

			final boolean isEmptyAccount = budgets.stream().map(budget -> budget.getBalances().get(account)).allMatch(balance -> balance == null || balance.getValue().compareTo(BigDecimal.ZERO) == 0);
